    const CONFIG = {

    const state = {
        speed: 0,
        currentGear: 1,
        angle: 0,
        pos: new THREE.Vector3(0, 0, 0),
        velocity: new THREE.Vector3(0, 0, 0),
        keys: { w: false, a: false, s: false, d: false, space: false },

    };

    function createTrackTexture() {
        const canvas = document.createElement('canvas');
        canvas.width = 1024;
        canvas.height = 1024;
        const ctx = canvas.getContext('2d');

        ctx.fillStyle = '#333';
        ctx.fillRect(0, 0, 1024, 1024);

        for (let i = 0; i < 60000; i++) {
            ctx.fillStyle = Math.random() > 0.5 ? '#3a3a3a' : '#282828';
            ctx.fillRect(Math.random() * 1024, Math.random() * 1024, 2, 2);
        }

        ctx.fillStyle = '#ffd700';
        ctx.fillRect(0, 0, 100, 1024);
        ctx.fillRect(924, 0, 100, 1024);

        ctx.fillStyle = '#cc0000';
        for (let i = 0; i < 1024; i += 64) {
            ctx.fillRect(0, i, 100, 32);
            ctx.fillRect(924, i, 100, 32);
        }

        const kSize = 50;
        for (let i = 0; i < 1024; i += 64) {
            ctx.fillStyle = '#cc0000';
            ctx.fillRect(100, i, kSize, 32);
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(100, i + 32, kSize, 32);
            ctx.fillStyle = '#cc0000';
            ctx.fillRect(1024 - 100 - kSize, i, kSize, 32);
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(1024 - 100 - kSize, i + 32, kSize, 32);
        }

        for (let i = 0; i < 1024; i += 80) {
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(512 - 4, i, 8, 40);
        }

        const texture = new THREE.CanvasTexture(canvas);
        texture.wrapS = THREE.RepeatWrapping;
        texture.wrapT = THREE.RepeatWrapping;
        return texture;
    }

    function createEnvironment() {
        const skyGeom = new THREE.SphereGeometry(2000, 32, 32);
        const skyMat = new THREE.MeshBasicMaterial({ color: SKY_COLOR, side: THREE.BackSide });
        const sky = new THREE.Mesh(skyGeom, skyMat);
        scene.add(sky);

        const geom = new THREE.PlaneGeometry(30000, 30000);
        const mat = new THREE.MeshLambertMaterial({ color: 0x1a472a });
        const mesh = new THREE.Mesh(geom, mat);
        mesh.rotation.x = -Math.PI / 2;
        mesh.position.y = -0.5;
        mesh.receiveShadow = true;
        scene.add(mesh);

        const treeGeom = new THREE.ConeGeometry(4, 12, 8);
        const treeMat = new THREE.MeshLambertMaterial({ color: 0x0d2b12 });

        const safeDistance = (CONFIG.trackWidth / 2) + 10;

        for (let i = 0; i < 800; i++) {
            const x = (Math.random() - 0.5) * 3000;
            const z = (Math.random() - 0.5) * 3000;

            let tooClose = false;
            for (let j = 0; j < state.trackPoints.length; j += 15) {
                const dx = x - state.trackPoints[j].x;
                const dz = z - state.trackPoints[j].z;
                if (dx * dx + dz * dz < safeDistance * safeDistance) {
                    tooClose = true;
                    break;
                }
            }
            if (tooClose) continue;

            const tree = new THREE.Mesh(treeGeom, treeMat);
            tree.position.set(x, 6, z);
            tree.castShadow = true;
            scene.add(tree);
        }
    }

    let trackCurve;

    function buildTrack() {
        // straight section first, then turns
        const points = [
            new THREE.Vector3(0, 0, 0),
            new THREE.Vector3(0, 0, -80),
            new THREE.Vector3(0, 0, -160),
            new THREE.Vector3(200, 0, -220),
            new THREE.Vector3(350, 0, -120),
            new THREE.Vector3(350, 0, 50),
            new THREE.Vector3(200, 0, 200),
            new THREE.Vector3(50, 0, 150),
            new THREE.Vector3(-100, 0, 200),
            new THREE.Vector3(-250, 0, 100),
            new THREE.Vector3(-200, 0, -50),
            new THREE.Vector3(-80, 0, -50)
        ];

        trackCurve = new THREE.CatmullRomCurve3(points, true);

        const divisions = 400;
        state.trackPoints = [];
        for (let i = 0; i < divisions; i++) {
            const t = i / divisions;
            const pt = trackCurve.getPointAt(t);
            state.trackPoints.push(pt);
            if (i % 10 === 0) state.checkpoints.push(pt);
        }

        const texture = createTrackTexture();
        texture.repeat.set(1, 60);

        const trackDivisions = 600;
        const verts = [];
        const uvs = [];
        const indices = [];

        const w = CONFIG.trackWidth / 2;
        const h = CONFIG.wallHeight;
        const wt = 1.0;

        const profile = [
            { x: -w - wt, y: h, u: 0.0 },
            { x: -w, y: h, u: 0.1 },
            { x: -w, y: 0, u: 0.15 },
            { x: w, y: 0, u: 0.85 },
            { x: w, y: h, u: 0.9 },
            { x: w + wt, y: h, u: 1.0 }
        ];

        const P = new THREE.Vector3();
        const T = new THREE.Vector3();
        const N = new THREE.Vector3(0, 1, 0);
        const B = new THREE.Vector3();
        const Temp = new THREE.Vector3();

        for (let i = 0; i <= trackDivisions; i++) {
            const t = i / trackDivisions;
            trackCurve.getPointAt(t % 1, P);
            trackCurve.getTangentAt(t % 1, T);

            B.crossVectors(T, N).normalize();

            for (let j = 0; j < profile.length; j++) {
                const prof = profile[j];

                Temp.copy(P);
                Temp.addScaledVector(B, prof.x);
                Temp.addScaledVector(N, prof.y);

                verts.push(Temp.x, Temp.y, Temp.z);
                uvs.push(prof.u, t * 60);
            }
        }

        const vertsPerStep = profile.length;
        for (let i = 0; i < trackDivisions; i++) {
            for (let j = 0; j < vertsPerStep - 1; j++) {
                const a = i * vertsPerStep + j;
                const b = i * vertsPerStep + j + 1;
                const c = (i + 1) * vertsPerStep + j;
                const d = (i + 1) * vertsPerStep + j + 1;

                indices.push(a, c, b);
                indices.push(b, c, d);
            }
        }

        const trackGeom = new THREE.BufferGeometry();
        trackGeom.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
        trackGeom.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
        trackGeom.setIndex(indices);
        trackGeom.computeVertexNormals();

        const trackMat = new THREE.MeshStandardMaterial({
            map: texture,
            roughness: 0.7,
            side: THREE.DoubleSide
        });

        const trackMesh = new THREE.Mesh(trackGeom, trackMat);
        trackMesh.position.y = 0.05;
        trackMesh.receiveShadow = true;
        scene.add(trackMesh);

        const startLineGeom = new THREE.PlaneGeometry(CONFIG.trackWidth - 2, 4);
        const startLineMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
        const startLine = new THREE.Mesh(startLineGeom, startLineMat);
        startLine.rotation.x = -Math.PI / 2;
        startLine.position.set(0, 0.07, 0);
        scene.add(startLine);
    }

    let carContainer;
    let carMesh;
    let frontWheels = [];

    function createCarMesh(colorHex) {
        const carGroup = new THREE.Group();

        const redMaterial = new THREE.MeshStandardMaterial({ color: colorHex, roughness: 0.2, metalness: 0.4 });
        const blackMaterial = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.8 });
        const darkMaterial = new THREE.MeshStandardMaterial({ color: 0x333333, roughness: 0.5 });

        const bodyGeom = new THREE.BoxGeometry(0.8, 0.4, 3.5);
        const body = new THREE.Mesh(bodyGeom, redMaterial);
        body.position.y = 0.5;
        body.castShadow = true;
        carGroup.add(body);

        const fWing = new THREE.Mesh(new THREE.BoxGeometry(2.4, 0.1, 0.6), darkMaterial);
        fWing.position.set(0, 0.2, 2.8);
        fWing.castShadow = true;
        carGroup.add(fWing);

        const wheelGeom = new THREE.CylinderGeometry(0.4, 0.4, 0.5, 24);
        wheelGeom.rotateZ(Math.PI / 2);

        function makeWheel(x, z, isFront) {
            const wheel = new THREE.Mesh(wheelGeom, blackMaterial);
            wheel.position.set(x, 0.4, z);
            wheel.castShadow = true;
            if (isFront) {
                const pivot = new THREE.Group();
                pivot.position.set(x, 0.4, z);
                wheel.position.set(0, 0, 0);
                pivot.add(wheel);
                carGroup.add(pivot);
            } else {
                carGroup.add(wheel);
            }
        }

        makeWheel(-1.0, 1.8, true);
        makeWheel(1.0, 1.8, true);
        makeWheel(-1.1, -1.5, false);
        makeWheel(1.1, -1.5, false);

        return carGroup;
    }

    // 2x2 grid in the straight section:
    // row 1 (front, nearer to turn): z = -10, lanes x = -4, 4
    // row 2 (back):                  z = -18, lanes x = -4, 4
    const GRID_POSITIONS = [
        { name: 'CAR 1', x: -4, z: -10 },  // AI 1
        { name: 'CAR 2', x:  4, z: -10 },  // AI 2
        { name: 'CAR 3', x: -4, z: -18 },  // AI 3
        { name: 'YOU',  x:  4, z: -18 }    // player
    ];

    function positionPlayerOnGrid() {
        const slot = GRID_POSITIONS[3]; // YOU
        carMesh.position.set(slot.x, 0.5, slot.z);
        state.angle = Math.PI; // face -Z
        carMesh.rotation.y = state.angle;
    }

    function createCar() {
        carContainer = createCarMesh(0xdc0000);
        scene.add(carContainer);
        carMesh = carContainer;

        frontWheels = [];
        carContainer.traverse(obj => {
            if (obj instanceof THREE.Group && obj.children.length === 1 && obj.children[0] instanceof THREE.Mesh) {
                frontWheels.push(obj);
            }
        });
    }

    function createAICars() {
        const colors = [0x0000ff, 0x00ff00, 0xffff00]; // blue, green, yellow
        const speeds = AI_SPEEDS[AI_DIFFICULTY] || AI_SPEEDS.medium;

        for (let i = 0; i < colors.length; i++) {
            const mesh = createCarMesh(colors[i]);
            scene.add(mesh);

            const slot = GRID_POSITIONS[i];
            mesh.position.set(slot.x, 0.5, slot.z);
            mesh.rotation.y = Math.PI; // face -Z like player

            state.aiCars.push({
                mesh,
                t: 0,                // progress along track
                speed: speeds[i] || speeds[speeds.length - 1],
                lateralOffset: slot.x, // used to keep different lanes
                lap: 1
            });
        }
    }

    function checkCollisionWalls() {
        let closestDistSq = Infinity;
        let closestPoint = null;

        for (let p of state.trackPoints) {
            const dSq = p.distanceToSquared(carMesh.position);
            if (dSq < closestDistSq) {
                closestDistSq = dSq;
                closestPoint = p;
            }
        }

        if (closestDistSq > (CONFIG.trackWidth / 2 - 1.5) ** 2) {
            const pushDir = new THREE.Vector3().subVectors(closestPoint, carMesh.position).normalize();
            carMesh.position.add(pushDir.multiplyScalar(0.5));
            state.speed *= 0.5;
        }
    }

    function checkCollisionCars() {
        const playerPos = carMesh.position;
        const tmp = new THREE.Vector3();
        const normal = new THREE.Vector3();
        const minDist = 2.2;

        state.aiCars.forEach(car => {
            tmp.copy(car.mesh.position);
            const dist = tmp.distanceTo(playerPos);
            if (dist < minDist) {
                normal.subVectors(playerPos, tmp).normalize();
                const overlap = minDist - dist;

                playerPos.addScaledVector(normal, overlap * 0.6);
                car.mesh.position.addScaledVector(normal, -overlap * 0.4);

                state.speed *= 0.4;
                car.speed *= 0.9;
            }
        });
    }

    function handleInput() {
        if (!state.gameActive) return;

        const gearLimitKmh = GEAR_LIMITS[state.currentGear];
        const gearLimitUnit = gearLimitKmh / 80;

        if (state.keys.w) {
            if (state.speed < gearLimitUnit) {
                const accelFactor = 1 - (Math.abs(state.speed) / (200 / 80));
                state.speed += CONFIG.accel * Math.max(0.2, accelFactor);
            }
        } else {
            state.speed *= CONFIG.friction;
        }

        if (state.keys.s) {
            if (state.speed > 0.1) {
                state.speed -= CONFIG.accel * 2.0;
            } else {
                state.speed -= CONFIG.accel * 0.5;
            }
        }

        if (state.keys.space) state.speed *= 0.85;

        const speedRatio = Math.min(1, Math.abs(state.speed) / (200 / 80));
        const steerSensitivity = CONFIG.turnSpeed * (1.2 - speedRatio * 0.5);
        const directionMult = state.speed < -0.1 ? -1 : 1;

        if (Math.abs(state.speed) > 0.05) {
            if (state.keys.a) {
                state.angle += steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = 0.4);
            } else if (state.keys.d) {
                state.angle -= steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = -0.4);
            } else {
                frontWheels.forEach(w => w.rotation.y = 0);
            }
        }

        if (state.speed < -0.6) state.speed = -0.6;
    }

    function updatePhysics() {
        if (!state.gameActive) return;

        state.velocity.x = Math.sin(state.angle) * state.speed;
        state.velocity.z = Math.cos(state.angle) * state.speed;

        carMesh.position.x += state.velocity.x;
        carMesh.position.z += state.velocity.z;
        carMesh.rotation.y = state.angle;

        checkCollisionWalls();
        checkCollisionCars();

        const carPos = carMesh.position;
        const currentCheck = state.checkpoints[state.nextCheckpoint];
        if (carPos.distanceTo(currentCheck) < 30) {
            state.nextCheckpoint = (state.nextCheckpoint + 1) % state.checkpoints.length;
            if (state.nextCheckpoint === 1) {
                const now = Date.now();
                if (state.lapStartTime > 0) {
                    state.lap++;
                    document.getElementById('lap-val').innerText = `${state.lap} / 5`;
                }
                state.lapStartTime = now;
            }
        }
    }

    function updateAICars() {
        if (!trackCurve) return;
        if (!state.gameActive) return;

        const up = new THREE.Vector3(0, 1, 0);
        const pos = new THREE.Vector3();
        const tangent = new THREE.Vector3();
        const side = new THREE.Vector3();

        state.aiCars.forEach(car => {
            const prevT = car.t;
            car.t = (car.t + car.speed) % 1;

            // lap counting
            if (prevT > 0.9 && car.t < 0.1) {
                car.lap += 1;
            }

            trackCurve.getPointAt(car.t, pos);
            trackCurve.getTangentAt(car.t, tangent);

            side.crossVectors(tangent, up).normalize();

            const finalPos = new THREE.Vector3().copy(pos)
                .addScaledVector(side, car.lateralOffset);
            finalPos.y = 0.5;

            car.mesh.position.copy(finalPos);
            car.mesh.rotation.y = Math.atan2(tangent.x, tangent.z);
        });
    }

    function updateCamera() {
        if (!carMesh) return;

        const relativeOffset = new THREE.Vector3(0.2, CONFIG.cameraHeight, -CONFIG.cameraDistance);
        const cameraOffset = relativeOffset.applyAxisAngle(new THREE.Vector3(0, 1, 0), state.angle);
        const targetPos = new THREE.Vector3(
            carMesh.position.x + cameraOffset.x,
            carMesh.position.y + cameraOffset.y,
            carMesh.position.z + cameraOffset.z
        );

        camera.position.lerp(targetPos, 0.4);
        const lookTarget = new THREE.Vector3(
            carMesh.position.x + Math.sin(state.angle) * 30,
            carMesh.position.y,
            carMesh.position.z + Math.cos(state.angle) * 30
        );
        camera.lookAt(lookTarget);
    }

    function getPlayerTrackIndex() {
        // find nearest track point index for ranking
        const pos = carMesh.position;
        let bestIdx = 0;
        let bestDist = Infinity;
        for (let i = 0; i < state.trackPoints.length; i++) {
            const d = state.trackPoints[i].distanceToSquared(pos);
            if (d < bestDist) {
                bestDist = d;
                bestIdx = i;
            }
        }
        return bestIdx;
    }

    function updateRankingHUD() {
        const listEl = document.getElementById('ranking-list');
        if (!trackCurve || !carMesh) return;

        const totalPoints = state.trackPoints.length || 1;

        const playerIndex = getPlayerTrackIndex();
        const cars = [];

        cars.push({
            name: 'YOU',
            lap: state.lap,
            progress: playerIndex / totalPoints
        });

        state.aiCars.forEach((car, i) => {
            const idx = Math.floor(car.t * totalPoints);
            cars.push({
                name: `CAR ${i + 1}`,
                lap: car.lap || 1,
                progress: idx / totalPoints
            });
        });

        cars.sort((a, b) => {
            if (b.lap !== a.lap) return b.lap - a.lap;
            return b.progress - a.progress;
        });

        listEl.innerHTML = '';
        cars.forEach((c, i) => {
            const row = document.createElement('div');
            const nameSpan = document.createElement('span');
            nameSpan.className = 'name';
            nameSpan.textContent = `${i + 1}. ${c.name}`;
            const lapSpan = document.createElement('span');
            lapSpan.textContent = `L${c.lap}`;
            row.appendChild(nameSpan);
            row.appendChild(lapSpan);
            listEl.appendChild(row);
        });
    }

    function updateHUD() {
        if (!state.gameActive) return;
        const visualSpeed = Math.abs(Math.floor(state.speed * 80));
        document.getElementById('speed-val').innerText = visualSpeed;

        let gearDisplay = state.currentGear;

        if (state.lapStartTime > 0) {
            const diff = Date.now() - state.lapStartTime;
            document.getElementById('time-val').innerText = formatTime(diff);
        }

        updateRankingHUD();
    }

    function formatTime(ms) {
        const min = Math.floor(ms / 60000);
        const sec = Math.floor((ms % 60000) / 1000);
        const mil = Math.floor((ms % 1000) / 100);
        return `${min < 10 ? '0' + min : min}:${sec < 10 ? '0' + sec : sec}.${mil}`;
    }

        if (e.key === 'ArrowDown') state.keys.s = true;

        if (e.key === 'ArrowRight') state.keys.d = true;

        if (key === 'e') {
            state.currentGear++;
            if (state.currentGear > 7) state.currentGear = 1;
        }

        if (key === 'r') {
            state.currentGear--;
            if (state.currentGear < 1) state.currentGear = 7;
        }

        if (state.keys.hasOwnProperty(key)) state.keys[key] = false;
        if (e.key === 'ArrowUp') state.keys.w = false;
        if (e.key === 'ArrowDown') state.keys.s = false;

    function startCountdown() {
        const loading = document.getElementById('loading');
        state.countdownActive = true;
        state.gameActive = false;

        let step = 3;
        loading.style.display = 'block';
        loading.innerText = `Difficulty: ${AI_DIFFICULTY.toUpperCase()}  |  ${step}`;

        const interval = setInterval(() => {
            step--;
            if (step > 0) {
                loading.innerText = `Difficulty: ${AI_DIFFICULTY.toUpperCase()}  |  ${step}`;
            } else if (step === 0) {
                loading.innerText = 'GO!';
            } else {
                clearInterval(interval);
                loading.style.display = 'none';
                state.gameActive = true;
                state.countdownActive = false;
                state.lapStartTime = Date.now();
            }
        }, 1000);
    }

    function init() {

        updateHUD();
        renderer.render(scene, camera);
    }

    init();

    function createTrackTexture() {
        const canvas = document.createElement('canvas');
        canvas.width = 1024;
        canvas.height = 1024;
        const ctx = canvas.getContext('2d');

        ctx.fillStyle = '#333';
        ctx.fillRect(0, 0, 1024, 1024);

        for (let i = 0; i < 60000; i++) {
            ctx.fillStyle = Math.random() > 0.5 ? '#3a3a3a' : '#282828';
            ctx.fillRect(Math.random() * 1024, Math.random() * 1024, 2, 2);

        ctx.fillStyle = '#ffd700';

        ctx.fillStyle = '#cc0000';

        }

        const kSize = 50;
        for (let i = 0; i < 1024; i += 64) {
            ctx.fillStyle = '#cc0000';
            ctx.fillRect(100, i, kSize, 32);
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(100, i + 32, kSize, 32);
            ctx.fillStyle = '#cc0000';
            ctx.fillRect(1024 - 100 - kSize, i, kSize, 32);
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(1024 - 100 - kSize, i + 32, kSize, 32);

            ctx.fillStyle = '#ffffff';

    function createEnvironment() {

    let trackCurve;

    function buildTrack() {
        // straight section first, then turns
        const points = [
            new THREE.Vector3(0, 0, 0),
            new THREE.Vector3(0, 0, -80),
            new THREE.Vector3(0, 0, -160),
            new THREE.Vector3(200, 0, -220),
            new THREE.Vector3(350, 0, -120),
            new THREE.Vector3(350, 0, 50),
            new THREE.Vector3(200, 0, 200),
            new THREE.Vector3(50, 0, 150),
            new THREE.Vector3(-100, 0, 200),
            new THREE.Vector3(-250, 0, 100),
            new THREE.Vector3(-200, 0, -50),
            new THREE.Vector3(-80, 0, -50)
        ];

        trackCurve = new THREE.CatmullRomCurve3(points, true);

        const divisions = 400;
        state.trackPoints = [];
        for (let i = 0; i < divisions; i++) {
            const t = i / divisions;
            const pt = trackCurve.getPointAt(t);
            state.trackPoints.push(pt);
            if (i % 10 === 0) state.checkpoints.push(pt);
        }

        const texture = createTrackTexture();
        texture.repeat.set(1, 60);

        const trackDivisions = 600;
        const verts = [];
        const uvs = [];
        const indices = [];

        const w = CONFIG.trackWidth / 2;
        const h = CONFIG.wallHeight;
        const wt = 1.0;

        const profile = [
            { x: -w - wt, y: h, u: 0.0 },
            { x: -w, y: h, u: 0.1 },
            { x: -w, y: 0, u: 0.15 },
            { x: w, y: 0, u: 0.85 },
            { x: w, y: h, u: 0.9 },
            { x: w + wt, y: h, u: 1.0 }
        ];

        const P = new THREE.Vector3();
        const T = new THREE.Vector3();
        const N = new THREE.Vector3(0, 1, 0);
        const B = new THREE.Vector3();
        const Temp = new THREE.Vector3();

        for (let i = 0; i <= trackDivisions; i++) {
            const t = i / trackDivisions;
            trackCurve.getPointAt(t % 1, P);
            trackCurve.getTangentAt(t % 1, T);

            B.crossVectors(T, N).normalize();

            for (let j = 0; j < profile.length; j++) {
                const prof = profile[j];

                Temp.copy(P);
                Temp.addScaledVector(B, prof.x);
                Temp.addScaledVector(N, prof.y);

                verts.push(Temp.x, Temp.y, Temp.z);
                uvs.push(prof.u, t * 60);
            }
        }

        const vertsPerStep = profile.length;
        for (let i = 0; i < trackDivisions; i++) {
            for (let j = 0; j < vertsPerStep - 1; j++) {
                const a = i * vertsPerStep + j;
                const b = i * vertsPerStep + j + 1;
                const c = (i + 1) * vertsPerStep + j;
                const d = (i + 1) * vertsPerStep + j + 1;

                indices.push(a, c, b);
                indices.push(b, c, d);
            }
        }

        const trackGeom = new THREE.BufferGeometry();
        trackGeom.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
        trackGeom.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
        trackGeom.setIndex(indices);
        trackGeom.computeVertexNormals();

        const trackMat = new THREE.MeshStandardMaterial({
            map: texture,
            roughness: 0.7,
            side: THREE.DoubleSide
        });

        const trackMesh = new THREE.Mesh(trackGeom, trackMat);
        trackMesh.position.y = 0.05;
        trackMesh.receiveShadow = true;
        scene.add(trackMesh);

        const startLineGeom = new THREE.PlaneGeometry(CONFIG.trackWidth - 2, 4);
        const startLineMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
        const startLine = new THREE.Mesh(startLineGeom, startLineMat);
        startLine.rotation.x = -Math.PI / 2;
        startLine.position.set(0, 0.07, 0);
        scene.add(startLine);
    }

    function createCarMesh(colorHex) {
        const carGroup = new THREE.Group();

        const redMaterial = new THREE.MeshStandardMaterial({ color: colorHex, roughness: 0.2, metalness: 0.4 });
        const blackMaterial = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.8 });
        const darkMaterial = new THREE.MeshStandardMaterial({ color: 0x333333, roughness: 0.5 });

        const bodyGeom = new THREE.BoxGeometry(0.8, 0.4, 3.5);
        const body = new THREE.Mesh(bodyGeom, redMaterial);
        body.position.y = 0.5;
        body.castShadow = true;
        carGroup.add(body);

        const fWing = new THREE.Mesh(new THREE.BoxGeometry(2.4, 0.1, 0.6), darkMaterial);
        fWing.position.set(0, 0.2, 2.8);
        fWing.castShadow = true;
        carGroup.add(fWing);

        const wheelGeom = new THREE.CylinderGeometry(0.4, 0.4, 0.5, 24);
        wheelGeom.rotateZ(Math.PI / 2);

        function makeWheel(x, z, isFront) {
            const wheel = new THREE.Mesh(wheelGeom, blackMaterial);
            wheel.position.set(x, 0.4, z);
            wheel.castShadow = true;
            if (isFront) {
                const pivot = new THREE.Group();
                pivot.position.set(x, 0.4, z);
                wheel.position.set(0, 0, 0);
                pivot.add(wheel);
                carGroup.add(pivot);
            } else {
                carGroup.add(wheel);
            }
        }

        makeWheel(-1.0, 1.8, true);
        makeWheel(1.0, 1.8, true);
        makeWheel(-1.1, -1.5, false);
        makeWheel(1.1, -1.5, false);

        return carGroup;
    }

    function createAICars() {
        const colors = [0x0000ff, 0x00ff00, 0xffff00]; // blue, green, yellow
        const speeds = AI_SPEEDS[AI_DIFFICULTY] || AI_SPEEDS.medium;

        for (let i = 0; i < colors.length; i++) {

            scene.add(mesh);

            mesh.position.set(slot.x, 0.5, slot.z);
            mesh.rotation.y = Math.PI;

            state.aiCars.push({
                mesh,
                t: 0,
                speed: speeds[i] || speeds[speeds.length - 1],
                lateralOffset: slot.x,
                lap: 1
            });
        }
    }

    function checkCollisionWalls() {
        let closestDistSq = Infinity;
        let closestPoint = null;

        for (let p of state.trackPoints) {
            const dSq = p.distanceToSquared(carMesh.position);
            if (dSq < closestDistSq) {
                closestDistSq = dSq;
                closestPoint = p;
            }
        }

        if (closestDistSq > (CONFIG.trackWidth / 2 - 1.5) ** 2) {
            const pushDir = new THREE.Vector3().subVectors(closestPoint, carMesh.position).normalize();
            carMesh.position.add(pushDir.multiplyScalar(0.5));
            state.speed *= 0.5;
        }
    }

    function checkCollisionCars() {
        const playerPos = carMesh.position;
        const tmp = new THREE.Vector3();
        const normal = new THREE.Vector3();
        const minDist = 2.2;

        state.aiCars.forEach(car => {
            tmp.copy(car.mesh.position);
            const dist = tmp.distanceTo(playerPos);
            if (dist < minDist) {
                normal.subVectors(playerPos, tmp).normalize();
                const overlap = minDist - dist;

                playerPos.addScaledVector(normal, overlap * 0.6);
                car.mesh.position.addScaledVector(normal, -overlap * 0.4);

                state.speed *= 0.4;
                car.speed *= 0.9;
            }
        });
    }

    function handleInput() {
        if (!state.gameActive) return;

        const gearLimitKmh = GEAR_LIMITS[state.currentGear];
        const gearLimitUnit = gearLimitKmh / 80;

        if (state.keys.w) {
            if (state.speed < gearLimitUnit) {
                const accelFactor = 1 - (Math.abs(state.speed) / (200 / 80));
                state.speed += CONFIG.accel * Math.max(0.2, accelFactor);
            }
        } else {
            state.speed *= CONFIG.friction;
        }

        if (state.keys.s) {
            if (state.speed > 0.1) {
                state.speed -= CONFIG.accel * 2.0;
            } else {
                state.speed -= CONFIG.accel * 0.5;
            }
        }

        if (state.keys.space) state.speed *= 0.85;

        const speedRatio = Math.min(1, Math.abs(state.speed) / (200 / 80));
        const steerSensitivity = CONFIG.turnSpeed * (1.2 - speedRatio * 0.5);
        const directionMult = state.speed < -0.1 ? -1 : 1;

        if (Math.abs(state.speed) > 0.05) {
            if (state.keys.a) {
                state.angle += steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = 0.4);
            } else if (state.keys.d) {
                state.angle -= steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = -0.4);
            } else {
                frontWheels.forEach(w => w.rotation.y = 0);
            }
        }

        if (state.speed < -0.6) state.speed = -0.6;
    }

    function updatePhysics() {
        if (!state.gameActive) return;

        state.velocity.x = Math.sin(state.angle) * state.speed;
        state.velocity.z = Math.cos(state.angle) * state.speed;

        carMesh.position.x += state.velocity.x;
        carMesh.position.z += state.velocity.z;
        carMesh.rotation.y = state.angle;

        checkCollisionWalls();
        checkCollisionCars();

        const carPos = carMesh.position;
        const currentCheck = state.checkpoints[state.nextCheckpoint];
        if (carPos.distanceTo(currentCheck) < 30) {
            state.nextCheckpoint = (state.nextCheckpoint + 1) % state.checkpoints.length;
            if (state.nextCheckpoint === 1) {

                if (state.lapStartTime > 0) {

    function updateAICars() {
        if (!trackCurve) return;
        if (!state.gameActive) return;

        const up = new THREE.Vector3(0, 1, 0);
        const pos = new THREE.Vector3();
        const tangent = new THREE.Vector3();
        const side = new THREE.Vector3();

        state.aiCars.forEach(car => {
            const prevT = car.t;
            car.t = (car.t + car.speed) % 1;

            if (prevT > 0.9 && car.t < 0.1) {
                car.lap += 1;
            }

            trackCurve.getPointAt(car.t, pos);
            trackCurve.getTangentAt(car.t, tangent);

            side.crossVectors(tangent, up).normalize();

            const finalPos = new THREE.Vector3().copy(pos)
                .addScaledVector(side, car.lateralOffset);
            finalPos.y = 0.5;

            car.mesh.position.copy(finalPos);
            car.mesh.rotation.y = Math.atan2(tangent.x, tangent.z);
        });
    }

    function buildTrack() {
        const points = [
            new THREE.Vector3(0, 0, 0),
            new THREE.Vector3(0, 0, -80),
            new THREE.Vector3(0, 0, -160),
            new THREE.Vector3(200, 0, -220),
            new THREE.Vector3(350, 0, -120),
            new THREE.Vector3(350, 0, 50),
            new THREE.Vector3(200, 0, 200),
            new THREE.Vector3(50, 0, 150),
            new THREE.Vector3(-100, 0, 200),
            new THREE.Vector3(-250, 0, 100),
            new THREE.Vector3(-200, 0, -50),
            new THREE.Vector3(-80, 0, -50)
        ];

        trackCurve = new THREE.CatmullRomCurve3(points, true);

        const divisions = 400;
        state.trackPoints = [];
        state.checkpoints = [];
        for (let i = 0; i < divisions; i++) {
            const t = i / divisions;
            const pt = trackCurve.getPointAt(t);
            state.trackPoints.push(pt);
            if (i % 10 === 0) state.checkpoints.push(pt);
        }

        const texture = createTrackTexture();
        texture.repeat.set(1, 60);

        const trackDivisions = 600;
        const verts = [];
        const uvs = [];
        const indices = [];

        const w = CONFIG.trackWidth / 2;
        const h = CONFIG.wallHeight;
        const wt = 1.0;

        const profile = [
            { x: -w - wt, y: h, u: 0.0 },
            { x: -w, y: h, u: 0.1 },
            { x: -w, y: 0, u: 0.15 },
            { x: w, y: 0, u: 0.85 },
            { x: w, y: h, u: 0.9 },
            { x: w + wt, y: h, u: 1.0 }
        ];

        const P = new THREE.Vector3();
        const T = new THREE.Vector3();
        const N = new THREE.Vector3(0, 1, 0);
        const B = new THREE.Vector3();
        const Temp = new THREE.Vector3();

        for (let i = 0; i <= trackDivisions; i++) {
            const t = i / trackDivisions;
            trackCurve.getPointAt(t % 1, P);
            trackCurve.getTangentAt(t % 1, T);

            B.crossVectors(T, N).normalize();

            for (let j = 0; j < profile.length; j++) {
                const prof = profile[j];

                Temp.copy(P);
                Temp.addScaledVector(B, prof.x);
                Temp.addScaledVector(N, prof.y);

                verts.push(Temp.x, Temp.y, Temp.z);
                uvs.push(prof.u, t * 60);
            }
        }

        const vertsPerStep = profile.length;
        for (let i = 0; i < trackDivisions; i++) {
            for (let j = 0; j < vertsPerStep - 1; j++) {
                const a = i * vertsPerStep + j;
                const b = i * vertsPerStep + j + 1;
                const c = (i + 1) * vertsPerStep + j;
                const d = (i + 1) * vertsPerStep + j + 1;

                indices.push(a, c, b);
                indices.push(b, c, d);
            }
        }

        const trackGeom = new THREE.BufferGeometry();
        trackGeom.setAttribute('position', new THREE.Float32BufferAttribute(verts, 3));
        trackGeom.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
        trackGeom.setIndex(indices);
        trackGeom.computeVertexNormals();

        const trackMat = new THREE.MeshStandardMaterial({
            map: texture,
            roughness: 0.7,
            side: THREE.DoubleSide
        });

        const trackMesh = new THREE.Mesh(trackGeom, trackMat);
        trackMesh.position.y = 0.05;
        trackMesh.receiveShadow = true;

        carMesh = carContainer;

        frontWheels = [];
        carContainer.traverse(obj => {
            if (obj instanceof THREE.Group && obj.children.length === 1 && obj.children[0] instanceof THREE.Mesh) {
                frontWheels.push(obj);
            }
        });
    }

    function createAICars() {
        const colors = [0x0000ff, 0x00ff00, 0xffff00];
        const speeds = AI_SPEEDS[AI_DIFFICULTY] || AI_SPEEDS.medium;

        for (let i = 0; i < colors.length; i++) {
            const mesh = createCarMesh(colors[i]);
            scene.add(mesh);

            const slot = GRID_POSITIONS[i];
            mesh.position.set(slot.x, 0.5, slot.z);
            mesh.rotation.y = Math.PI;

            state.aiCars.push({
                mesh,
                t: 0,
                speed: speeds[i] || speeds[speeds.length - 1],
                lateralOffset: slot.x,
                lap: 0
            });
        }
    }

    function checkCollisionWalls() {
        let closestDistSq = Infinity;
        let closestPoint = null;

        for (let p of state.trackPoints) {
            const dSq = p.distanceToSquared(carMesh.position);
            if (dSq < closestDistSq) {
                closestDistSq = dSq;
                closestPoint = p;
            }
        }

        if (closestDistSq > (CONFIG.trackWidth / 2 - 1.5) ** 2) {
            const pushDir = new THREE.Vector3().subVectors(closestPoint, carMesh.position).normalize();
            carMesh.position.add(pushDir.multiplyScalar(0.5));
            state.speed *= 0.5;

            state.currentGear = 1;
            state.gearLockUntil = Date.now() + 1250;
        }
    }

    function checkCollisionCars() {
        const playerPos = carMesh.position;
        const tmp = new THREE.Vector3();
        const normal = new THREE.Vector3();
        const minDist = 2.2;

        state.aiCars.forEach(car => {
            tmp.copy(car.mesh.position);
            const dist = tmp.distanceTo(playerPos);
            if (dist < minDist) {
                normal.subVectors(playerPos, tmp).normalize();
                const overlap = minDist - dist;

                playerPos.addScaledVector(normal, overlap * 0.6);
                car.mesh.position.addScaledVector(normal, -overlap * 0.4);

                state.speed *= 0.4;
                car.speed *= 0.9;
            }
        });
    }

    function handleInput() {
        if (!state.gameActive) return;

        const gearLimitKmh = GEAR_LIMITS[state.currentGear];
        const gearLimitUnit = gearLimitKmh / 80;

        if (state.keys.w) {
            if (state.speed < gearLimitUnit) {
                const accelFactor = 1 - (Math.abs(state.speed) / (200 / 80));
                state.speed += CONFIG.accel * Math.max(0.2, accelFactor);
            }
        } else {
            state.speed *= CONFIG.friction;
        }

        if (state.keys.s) {
            if (state.speed > 0.1) {
                state.speed -= CONFIG.accel * 2.0;
            } else {
                state.speed -= CONFIG.accel * 0.5;
            }
        }

        if (state.keys.space) state.speed *= 0.85;

        const speedRatio = Math.min(1, Math.abs(state.speed) / (200 / 80));
        const steerSensitivity = CONFIG.turnSpeed * (1.2 - speedRatio * 0.5);
        const directionMult = state.speed < -0.1 ? -1 : 1;

        if (Math.abs(state.speed) > 0.05) {
            if (state.keys.a) {
                state.angle += steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = 0.4);
            } else if (state.keys.d) {
                state.angle -= steerSensitivity * directionMult;
                frontWheels.forEach(w => w.rotation.y = -0.4);
            } else {
                frontWheels.forEach(w => w.rotation.y = 0);
            }
        }

        if (state.speed < -0.6) state.speed = -0.6;
    }

    function maybeFinishRace(winnerName) {
        if (state.raceFinished) return;
        state.raceFinished = true;
        state.winnerName = winnerName;
        state.gameActive = false;

        const loading = document.getElementById('loading');
        loading.style.display = 'block';
        loading.innerText = `${winnerName} WINS!\nRace finished after ${CONFIG.lapsToWin} laps.`;
    }

    function updatePhysics() {
        if (!state.gameActive) return;

        state.velocity.x = Math.sin(state.angle) * state.speed;
        state.velocity.z = Math.cos(state.angle) * state.speed;

        carMesh.position.x += state.velocity.x;
        carMesh.position.z += state.velocity.z;
        carMesh.rotation.y = state.angle;

        checkCollisionWalls();
        checkCollisionCars();

        const carPos = carMesh.position;
        const currentCheck = state.checkpoints[state.nextCheckpoint];
        if (carPos.distanceTo(currentCheck) < 30) {
            state.nextCheckpoint = (state.nextCheckpoint + 1) % state.checkpoints.length;
            // we just wrapped back to start/finish when nextCheckpoint became 0
            if (state.nextCheckpoint === 0) {
                const now = Date.now();
                if (state.lapStartTime > 0) {
                    state.lap++;
                    document.getElementById('lap-val').innerText =
                        `${state.lap} / ${CONFIG.lapsToWin}`;
                    if (state.lap >= CONFIG.lapsToWin) {
                        maybeFinishRace('YOU');
                    }
                }
                state.lapStartTime = now;
            }
        }
    }

    function updateAICars() {
        if (!trackCurve) return;
        if (!state.gameActive) return;

        const up = new THREE.Vector3(0, 1, 0);
        const pos = new THREE.Vector3();
        const tangent = new THREE.Vector3();
        const side = new THREE.Vector3();

        state.aiCars.forEach((car, index) => {
            const prevT = car.t;
            car.t = (car.t + car.speed) % 1;

            if (prevT > 0.9 && car.t < 0.1) {
                car.lap++;
                if (car.lap >= CONFIG.lapsToWin) {
                    maybeFinishRace(`CAR ${index + 1}`);
                }
            }

            trackCurve.getPointAt(car.t, pos);
            trackCurve.getTangentAt(car.t, tangent);

            side.crossVectors(tangent, up).normalize();

            const finalPos = new THREE.Vector3().copy(pos)
                .addScaledVector(side, car.lateralOffset);
            finalPos.y = 0.5;

            car.mesh.position.copy(finalPos);
            car.mesh.rotation.y = Math.atan2(tangent.x, tangent.z);
        });
    }

    function updateCamera() {
        if (!carMesh) return;

        const relativeOffset = new THREE.Vector3(0.2, CONFIG.cameraHeight, -CONFIG.cameraDistance);
        const cameraOffset = relativeOffset.applyAxisAngle(new THREE.Vector3(0, 1, 0), state.angle);
        const targetPos = new THREE.Vector3(
            carMesh.position.x + cameraOffset.x,
            carMesh.position.y + cameraOffset.y,
            carMesh.position.z + cameraOffset.z
        );

        camera.position.lerp(targetPos, 0.4);
        const lookTarget = new THREE.Vector3(
            carMesh.position.x + Math.sin(state.angle) * 30,
            carMesh.position.y,
            carMesh.position.z + Math.cos(state.angle) * 30
        );
        camera.lookAt(lookTarget);
    }

    function getNearestTrackIndexForPosition(pos) {
        let bestIdx = 0;
        let bestDist = Infinity;
        for (let i = 0; i < state.trackPoints.length; i++) {
            const d = state.trackPoints[i].distanceToSquared(pos);
            if (d < bestDist) {
                bestDist = d;
                bestIdx = i;
            }
        }
        return bestIdx;
    }

    function updateRankingHUD() {
        const listEl = document.getElementById('ranking-list');
        if (!trackCurve || !carMesh) return;

        const totalPoints = state.trackPoints.length || 1;
        const cars = [];

        const playerIndex = getNearestTrackIndexForPosition(carMesh.position);
        cars.push({
            name: 'YOU',
            lap: state.lap,
            progress: playerIndex / totalPoints
        });

        state.aiCars.forEach((car, i) => {
            const idx = getNearestTrackIndexForPosition(car.mesh.position);
            cars.push({
                name: `CAR ${i + 1}`,
                lap: car.lap || 0,
                progress: idx / totalPoints
            });
        });

        // Real race order: more laps first, then distance
        cars.sort((a, b) => {
            if (b.lap !== a.lap) return b.lap - a.lap;
            return b.progress - a.progress;
        });

        listEl.innerHTML = '';
        cars.forEach((c, i) => {
            const row = document.createElement('div');
            const nameSpan = document.createElement('span');
            nameSpan.className = 'name';
            nameSpan.textContent = `${i + 1}. ${c.name}`;
            const lapSpan = document.createElement('span');
            lapSpan.textContent = `L${c.lap}`;
            row.appendChild(nameSpan);
            row.appendChild(lapSpan);
            listEl.appendChild(row);
        });
    }

    function updateHUD() {
        if (!state.gameActive && !state.raceFinished) return;
        const visualSpeed = Math.abs(Math.floor(state.speed * 80));
        document.getElementById('speed-val').innerText = visualSpeed;

        let gearDisplay = state.currentGear;
        if (state.speed < -0.1) gearDisplay = "R";
        document.getElementById('gear').innerText = gearDisplay;

        if (state.lapStartTime > 0 && !state.raceFinished) {
            const diff = Date.now() - state.lapStartTime;
            document.getElementById('time-val').innerText = formatTime(diff);
        }

        updateRankingHUD();
    }

    function formatTime(ms) {
        const min = Math.floor(ms / 60000);
        const sec = Math.floor((ms % 60000) / 1000);
        const mil = Math.floor((ms % 1000) / 100);
        return `${min < 10 ? '0' + min : min}:${sec < 10 ? '0' + sec : sec}.${mil}`;
    }

    window.addEventListener('keydown', (e) => {
        const key = e.key.toLowerCase();
        if (state.keys.hasOwnProperty(key)) state.keys[key] = true;
        if (e.key === 'ArrowUp') state.keys.w = true;
        if (e.key === 'ArrowDown') state.keys.s = true;
        if (e.key === 'ArrowLeft') state.keys.a = true;
        if (e.key === 'ArrowRight') state.keys.d = true;
        if (e.code === 'Space') state.keys.space = true;
    });

    window.addEventListener('keyup', (e) => {
        const key = e.key.toLowerCase();

        const canShift = !state.gearLockUntil || Date.now() > state.gearLockUntil;

        if (canShift && key === 'e') {
            state.currentGear++;
            if (state.currentGear > 7) state.currentGear = 1;
        }

        if (canShift && key === 'r') {
            state.currentGear--;
            if (state.currentGear < 1) state.currentGear = 7;
        }

        if (state.keys.hasOwnProperty(key)) state.keys[key] = false;
        if (e.key === 'ArrowUp') state.keys.w = false;
        if (e.key === 'ArrowDown') state.keys.s = false;
        if (e.key === 'ArrowLeft') state.keys.a = false;
        if (e.key === 'ArrowRight') state.keys.d = false;
        if (e.code === 'Space') state.keys.space = false;
    });

    window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    function startCountdown() {
        const loading = document.getElementById('loading');
        state.countdownActive = true;
        state.gameActive = false;
        state.raceFinished = false;
        state.lap = 0;
        state.nextCheckpoint = 1;
        document.getElementById('lap-val').innerText = `0 / ${CONFIG.lapsToWin}`;

        state.aiCars.forEach(car => { car.lap = 0; });

        let step = 3;
        loading.style.display = 'block';
        loading.innerText = `Difficulty: ${AI_DIFFICULTY.toUpperCase()}\nRace to ${CONFIG.lapsToWin} laps\n${step}`;

        const interval = setInterval(() => {
            step--;
            if (step > 0) {
                loading.innerText = `Difficulty: ${AI_DIFFICULTY.toUpperCase()}\nRace to ${CONFIG.lapsToWin} laps\n${step}`;
            } else if (step === 0) {
                loading.innerText = 'GO!';
            } else {
                clearInterval(interval);
                loading.style.display = 'none';
                state.gameActive = true;
                state.countdownActive = false;
                state.lapStartTime = Date.now();
            }
        }, 1000);
    }

    function init() {
        const diff = prompt("AI difficulty: easy, medium, hard", AI_DIFFICULTY);
        if (diff && AI_SPEEDS[diff]) {
            AI_DIFFICULTY = diff;
        }

        buildTrack();
        createEnvironment();
        createCar();
        positionPlayerOnGrid();
        createAICars();

        startCountdown();
        animate();
    }

    function animate() {
        requestAnimationFrame(animate);
        handleInput();
        updatePhysics();
        updateAICars();
        updateCamera();
        updateHUD();
        renderer.render(scene, camera);
    }

    init();
</script>
